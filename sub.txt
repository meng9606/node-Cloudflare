/**
 * Sub Aggregator - Cloudflare Worker
 * 聚合订阅生成器：支持单节点 + 远程订阅源实时聚合
 * 自动识别 Clash/OpenClash 与 Shadowrocket/V2ray 等客户端，分别输出 YAML / Base64
 * 数据存储：Cloudflare KV（绑定名 SUB_KV）
 * 环境变量：ADMIN_USERNAME / ADMIN_PASSWORD
 *
 * 支持的节点协议：
 *   vless / vmess / trojan / ss(含 plugin) / ssr / hysteria / hysteria2(含端口跳跃)
 *   tuic / anytls / wireguard / masque / mieru / socks5
 *   naive、juicity：可保存并输出到 Base64 订阅，但 Clash 内核无对应类型，输出 YAML 时自动跳过
 * 传输层：tcp / ws / grpc / h2 / httpupgrade / xhttp
 *
 * 导入支持：换行、\r\n、空格分隔、字面 "\n"、整段 Base64（含折行）
 */

// ============================== 工具函数 ==============================

function b64EncodeUtf8(str) {
  const bytes = new TextEncoder().encode(str);
  let bin = '';
  bytes.forEach((b) => (bin += String.fromCharCode(b)));
  return btoa(bin);
}

function b64DecodeUtf8(str) {
  const bin = atob(fixB64(str));
  const bytes = Uint8Array.from(bin, (c) => c.charCodeAt(0));
  return new TextDecoder().decode(bytes);
}

function fixB64(s) {
  s = String(s).replace(/\s+/g, '').replace(/-/g, '+').replace(/_/g, '/');
  while (s.length % 4) s += '=';
  return s;
}

function looksLikeBase64(s) {
  const t = String(s).replace(/\s/g, '');
  if (t.length < 8) return false;
  return /^[A-Za-z0-9+/_-]+={0,2}$/.test(t);
}

function splitFirst(s, sep) {
  const i = s.indexOf(sep);
  if (i < 0) return [s, ''];
  return [s.slice(0, i), s.slice(i + 1)];
}

function splitLast(s, sep) {
  const i = s.lastIndexOf(sep);
  if (i < 0) return [s, ''];
  return [s.slice(0, i), s.slice(i + 1)];
}

function safeDecode(s) {
  try {
    return decodeURIComponent(s);
  } catch (e) {
    return s;
  }
}

function hostOf(u) {
  return u.hostname.replace(/^\[|\]$/g, '');
}

function listOf(s) {
  return s ? String(s).split(',').map((x) => x.trim()).filter(Boolean) : [];
}

function uid(len = 12) {
  const bytes = crypto.getRandomValues(new Uint8Array(len));
  return Array.from(bytes, (b) => b.toString(16).padStart(2, '0')).join('').slice(0, len);
}

function randomToken(len = 28) {
  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';
  const bytes = crypto.getRandomValues(new Uint8Array(len));
  return Array.from(bytes, (b) => chars[b % chars.length]).join('');
}

function jsonResponse(obj, status = 200, extraHeaders = {}) {
  return new Response(JSON.stringify(obj), {
    status,
    headers: { 'Content-Type': 'application/json; charset=utf-8', ...extraHeaders },
  });
}

function parseCookies(req) {
  const header = req.headers.get('Cookie') || '';
  const out = {};
  header.split(';').forEach((p) => {
    const [k, ...v] = p.trim().split('=');
    if (!k) return;
    out[k] = decodeURIComponent(v.join('='));
  });
  return out;
}

// 从任意文本中提取节点链接：换行 / \r\n / 空格分隔 / 字面 "\n" / 整段 Base64（含折行）
function extractNodeLinks(text) {
  const prep = (t) => String(t || '').replace(/\\n/g, '\n').replace(/\r\n?/g, '\n').trim();
  const alt = NODE_SCHEMES.map((s) => s.replace(/[+.]/g, '\\$&')).join('|');
  const splitRe = new RegExp('\\s+(?=(?:' + alt + ')://)', 'i');
  const startRe = new RegExp('^(?:' + alt + ')://', 'i');
  const pick = (t) => {
    const out = [];
    for (const line of t.split('\n')) {
      for (const seg of line.trim().split(splitRe)) {
        const s = seg.trim();
        if (startRe.test(s)) out.push(s);
      }
    }
    return out;
  };
  const t = prep(text);
  let links = pick(t);
  if (!links.length) {
    const compact = t.replace(/\s+/g, '');
    if (looksLikeBase64(compact)) {
      try {
        links = pick(prep(b64DecodeUtf8(compact)));
      } catch (e) {
        /* ignore */
      }
    }
  }
  return links;
}

// ============================== YAML 序列化（受控子集） ==============================

function yamlScalar(v) {
  if (v === null || v === undefined) return 'null';
  if (typeof v === 'number') return String(v);
  if (typeof v === 'boolean') return v ? 'true' : 'false';
  let s = String(v);
  if (s === '') return "''";
  const needsQuote =
    /[:#\[\]{}",\r\n]/.test(s) ||
    /^[\s]/.test(s) ||
    /[\s]$/.test(s) ||
    /^(true|false|null|~|yes|no)$/i.test(s) ||
    /^-?\d+(\.\d+)?$/.test(s) ||
    s.startsWith('-') ||
    s.startsWith('*') ||
    s.startsWith('&') ||
    s.startsWith('!') ||
    s.startsWith('>') ||
    s.startsWith('|') ||
    s.startsWith('%') ||
    s.startsWith('@') ||
    s.startsWith('`');
  if (needsQuote) {
    return `"${s.replace(/\\/g, '\\\\').replace(/"/g, '\\"').replace(/\r/g, '\\r').replace(/\n/g, '\\n')}"`;
  }
  return s;
}

function yamlDump(value, indent = 0) {
  const pad = '  '.repeat(indent);
  if (Array.isArray(value)) {
    if (value.length === 0) return pad + '[]\n';
    let out = '';
    for (const item of value) {
      if (item !== null && typeof item === 'object') {
        const inner = yamlDump(item, indent + 1);
        const lines = inner.split('\n').filter((l) => l.length > 0);
        lines.forEach((line, i) => {
          if (i === 0) {
            out += pad + '- ' + line.slice(pad.length + 2) + '\n';
          } else {
            out += line + '\n';
          }
        });
      } else {
        out += pad + '- ' + yamlScalar(item) + '\n';
      }
    }
    return out;
  }
  if (value !== null && typeof value === 'object') {
    let out = '';
    for (const [k, v] of Object.entries(value)) {
      if (v === undefined) continue;
      if (v !== null && typeof v === 'object' && !Array.isArray(v) && Object.keys(v).length === 0) continue;
      if (Array.isArray(v) && v.length === 0) continue;
      if (v !== null && typeof v === 'object') {
        out += pad + k + ':\n' + yamlDump(v, indent + 1);
      } else {
        out += pad + k + ': ' + yamlScalar(v) + '\n';
      }
    }
    return out;
  }
  return pad + yamlScalar(value) + '\n';
}

// ============================== 节点 URI 解析 ==============================

function safeUrlParse(raw) {
  return new URL(raw);
}

// wireguard / masque 的私钥可能含未编码的 / + =，先统一编码再交给 URL 解析
function encodeKeyUserinfo(raw) {
  return raw.replace(/^([a-z0-9]+):\/\/([^@]+)@/i, (m, scheme, key) =>
    scheme + '://' + encodeURIComponent(safeDecode(key)) + '@'
  );
}

function parseVless(raw) {
  const u = safeUrlParse(raw);
  const p = u.searchParams;
  const host = hostOf(u);
  const name = safeDecode(u.hash.replace(/^#/, '')) || host;
  let network = (p.get('type') || 'tcp').toLowerCase();
  if (network === 'raw') network = 'tcp';
  return {
    protocol: 'vless',
    name,
    server: host,
    port: parseInt(u.port, 10),
    uuid: safeDecode(u.username),
    security: p.get('security') || 'none',
    sni: p.get('sni') || p.get('host') || host,
    network,
    wsPath: p.get('path') ? safeDecode(p.get('path')) : '/',
    wsHost: p.get('host') || p.get('sni') || host,
    grpcServiceName: p.get('serviceName') || p.get('path') || '',
    xhttpMode: p.get('mode') || '',
    flow: p.get('flow') || '',
    fp: p.get('fp') || '',
    pbk: p.get('pbk') || '',
    sid: p.get('sid') || '',
    alpn: p.get('alpn') ? safeDecode(p.get('alpn')) : '',
    encryption: p.get('encryption') || '',
    insecure: p.get('insecure') === '1' || p.get('allowInsecure') === '1',
  };
}

function parseVmess(raw) {
  const b64 = raw.replace(/^vmess:\/\//i, '');
  const json = JSON.parse(b64DecodeUtf8(b64));
  const tls = String(json.tls || '').toLowerCase();
  return {
    protocol: 'vmess',
    name: json.ps || json.add || 'vmess',
    server: json.add,
    port: parseInt(json.port, 10),
    uuid: json.id,
    aid: parseInt(json.aid || 0, 10),
    cipher: json.scy || 'auto',
    network: json.net || 'tcp',
    tlsEnabled: tls === 'tls' || tls === 'reality',
    sni: json.sni || json.host || json.add,
    wsPath: json.path || '/',
    wsHost: json.host || json.add,
    grpcServiceName: json.path || '',
    alpn: json.alpn || '',
    fp: json.fp || '',
    insecure: json.allowInsecure === true || json.allowInsecure === 1 || json.allowInsecure === '1',
  };
}

function parseSS(raw) {
  let rest = raw.replace(/^ss:\/\//i, '');
  let name = '';
  const hashIdx = rest.indexOf('#');
  if (hashIdx >= 0) {
    name = safeDecode(rest.slice(hashIdx + 1));
    rest = rest.slice(0, hashIdx);
  }
  let query = '';
  const qIdx = rest.indexOf('?');
  if (qIdx >= 0) {
    query = rest.slice(qIdx + 1);
    rest = rest.slice(0, qIdx);
  }
  const plugin = query ? new URLSearchParams(query).get('plugin') || '' : '';

  let method, password, server, port;
  const atIdx = rest.lastIndexOf('@');
  if (atIdx >= 0) {
    // SIP002: base64(method:password)@host:port
    const userinfo = rest.slice(0, atIdx);
    const hostport = rest.slice(atIdx + 1).replace(/\/+$/, '');
    let decoded;
    try {
      decoded = b64DecodeUtf8(userinfo);
      if (!decoded.includes(':')) throw new Error('not method:password');
    } catch (e) {
      decoded = safeDecode(userinfo);
    }
    [method, password] = splitFirst(decoded, ':');
    [server, port] = splitLast(hostport, ':');
  } else {
    // legacy: base64(method:password@host:port)
    const decoded = b64DecodeUtf8(rest);
    const at2 = decoded.lastIndexOf('@');
    const userinfo = decoded.slice(0, at2);
    const hostport = decoded.slice(at2 + 1);
    [method, password] = splitFirst(userinfo, ':');
    [server, port] = splitLast(hostport, ':');
  }
  server = String(server).replace(/^\[|\]$/g, '');
  return {
    protocol: 'ss',
    name: name || server,
    server,
    port: parseInt(port, 10),
    method,
    password,
    plugin,
  };
}

function parseSSR(raw) {
  const body = raw.replace(/^ssr:\/\//i, '').split('#')[0];
  const decoded = b64DecodeUtf8(body);
  const qi = decoded.indexOf('/?');
  const main = qi >= 0 ? decoded.slice(0, qi) : decoded;
  const q = {};
  if (qi >= 0) {
    decoded
      .slice(qi + 2)
      .split('&')
      .forEach((kv) => {
        const i = kv.indexOf('=');
        if (i > 0) q[kv.slice(0, i)] = kv.slice(i + 1);
      });
  }
  const parts = main.split(':');
  if (parts.length < 6) throw new Error('SSR 格式不正确');
  const passB64 = parts.pop();
  const obfs = parts.pop();
  const method = parts.pop();
  const protocol = parts.pop();
  const port = parts.pop();
  const server = parts.join(':').replace(/^\[|\]$/g, '');
  const dec = (k) => {
    if (!q[k]) return '';
    try {
      return b64DecodeUtf8(q[k]);
    } catch (e) {
      return '';
    }
  };
  return {
    protocol: 'ssr',
    name: dec('remarks') || server,
    server,
    port: parseInt(port, 10),
    password: b64DecodeUtf8(passB64),
    method,
    ssrProtocol: protocol,
    obfs,
    obfsParam: dec('obfsparam'),
    protocolParam: dec('protoparam'),
  };
}

function parseWireguard(raw) {
  const u = safeUrlParse(encodeKeyUserinfo(raw));
  const p = u.searchParams;
  const host = hostOf(u);
  const name = safeDecode(u.hash.replace(/^#/, '')) || host;
  let ip = '';
  let ipv6 = '';
  listOf(p.get('address') || p.get('ip')).forEach((a) => {
    const addr = a.replace(/\/\d+$/, '');
    if (addr.includes(':')) ipv6 = ipv6 || addr;
    else ip = ip || addr;
  });
  const reserved = listOf(p.get('reserved'))
    .map(Number)
    .filter((x) => !Number.isNaN(x));
  return {
    protocol: 'wireguard',
    name,
    server: host,
    port: parseInt(u.port, 10) || 51820,
    privateKey: safeDecode(u.username),
    publicKey: p.get('publickey') || p.get('public-key') || p.get('peer_public_key') || '',
    presharedKey: p.get('presharedkey') || p.get('pre-shared-key') || '',
    ip,
    ipv6,
    reserved,
    mtu: parseInt(p.get('mtu'), 10) || 0,
  };
}

// mierus://用户:密码@host?port=2999&protocol=TCP&multiplexing=MULTIPLEXING_LOW&profile=名称#名称
// port 也可以是端口段，如 2090-2099（可重复出现，取第一个）
function parseMieru(raw) {
  const normalized = raw.replace(/^mieru:\/\//i, 'mierus://');
  const u = safeUrlParse(normalized);
  const p = u.searchParams;
  const host = hostOf(u);
  const portStr = p.get('port') || u.port || '';
  const first = portStr.split(/[,]/)[0];
  const portNum = parseInt(first, 10);
  const isRange = /^\d+-\d+$/.test(first);
  return {
    protocol: 'mieru',
    name: safeDecode(u.hash.replace(/^#/, '')) || p.get('profile') || host,
    server: host,
    port: portNum,
    portRange: isRange ? first : '',
    username: safeDecode(u.username),
    password: safeDecode(u.password),
    transport: (p.get('protocol') || 'TCP').toUpperCase(),
    multiplexing: p.get('multiplexing') || '',
  };
}

// socks5://user:pass@host:port  或  socks5://base64(user:pass)@host:port
function parseSocks(raw) {
  const u = safeUrlParse(raw);
  const host = hostOf(u);
  let user = safeDecode(u.username);
  let pass = safeDecode(u.password);
  if (user && !pass) {
    try {
      const d = b64DecodeUtf8(user);
      if (d.includes(':')) [user, pass] = splitFirst(d, ':');
    } catch (e) {
      /* 不是 base64 */
    }
  }
  return {
    protocol: 'socks5',
    name: safeDecode(u.hash.replace(/^#/, '')) || host,
    server: host,
    port: parseInt(u.port, 10) || 1080,
    username: user,
    password: pass,
  };
}

function detectProtocol(raw) {
  const m = raw.trim().match(/^([a-z0-9+.-]+):\/\//i);
  if (!m) return null;
  return SCHEME_MAP[m[1].toLowerCase()] || null;
}

function parseNodeUri(raw) {
  raw = raw.trim();
  const protocol = detectProtocol(raw);
  if (!protocol) throw new Error('无法识别的协议: ' + raw.slice(0, 30));
  const parsers = {
    vless: parseVless,
    vmess: parseVmess,
    trojan: parseTrojan,
    ss: parseSS,
    ssr: parseSSR,
    hysteria2: parseHysteria2,
    hysteria: parseHysteria,
    tuic: parseTuic,
    anytls: parseAnytls,
    wireguard: parseWireguard,
    masque: parseMasque,
    mieru: parseMieru,
    socks5: parseSocks,
    naive: parseNaive,
    juicity: parseJuicity,
  };
  const parsed = parsers[protocol](raw);
  if (!parsed.server || !parsed.port) throw new Error('缺少 server/port: ' + raw.slice(0, 30));
  parsed.raw = raw;
  return parsed;
}

// ============================== Clash proxy 对象生成 ==============================

function uniqueName(base, usedNames) {
  let name = base || 'node';
  let i = 2;
  while (usedNames.has(name)) {
    name = `${base} ${i}`;
    i++;
  }
  usedNames.add(name);
  return name;
}

function applyTransport(proxy, n) {
  const net = n.network;
  if (net === 'ws') {
    proxy['ws-opts'] = { path: n.wsPath, headers: { Host: n.wsHost } };
  } else if (net === 'httpupgrade') {
    proxy.network = 'ws';
    proxy['ws-opts'] = { path: n.wsPath, headers: { Host: n.wsHost }, 'v2ray-http-upgrade': true };
  } else if (net === 'grpc') {
    proxy['grpc-opts'] = { 'grpc-service-name': n.grpcServiceName };
  } else if (net === 'h2') {
    proxy['h2-opts'] = { host: [n.wsHost], path: n.wsPath };
  } else if (net === 'xhttp') {
    proxy['xhttp-opts'] = { path: n.wsPath, host: n.wsHost, mode: n.xhttpMode || 'auto' };
  }
}

function nodeToClashProxy(n) {
  const proxy = { name: n.name, server: n.server, port: n.port, udp: true };
  const alpn = listOf(n.alpn);
  switch (n.protocol) {
    case 'vless':
      proxy.type = 'vless';
      proxy.uuid = n.uuid;
      proxy.tls = n.security === 'tls' || n.security === 'reality';
      proxy.network = n.network === 'httpupgrade' ? 'ws' : n.network;
      proxy['skip-cert-verify'] = !!n.insecure;
      if (proxy.tls) proxy.servername = n.sni;
      if (proxy.tls && alpn.length) proxy.alpn = alpn;
      if (n.flow) proxy.flow = n.flow;
      if (n.fp) proxy['client-fingerprint'] = n.fp;
      if (n.encryption && n.encryption !== 'none') proxy.encryption = n.encryption;
      if (n.security === 'reality') {
        proxy['reality-opts'] = { 'public-key': n.pbk, 'short-id': n.sid };
      }
      applyTransport(proxy, n);
      break;
    case 'vmess':
      proxy.type = 'vmess';
      proxy.uuid = n.uuid;
      proxy.alterId = n.aid;
      proxy.cipher = n.cipher;
      proxy.tls = !!n.tlsEnabled;
      proxy.network = n.network === 'httpupgrade' ? 'ws' : n.network;
      if (proxy.tls) {
        proxy.servername = n.sni;
        proxy['skip-cert-verify'] = !!n.insecure;
        if (alpn.length) proxy.alpn = alpn;
        if (n.fp) proxy['client-fingerprint'] = n.fp;
      }
      applyTransport(proxy, n);
      break;
    case 'trojan':
      proxy.type = 'trojan';
      proxy.password = n.password;
      proxy.sni = n.sni;
      proxy['skip-cert-verify'] = !!n.insecure;
      if (alpn.length) proxy.alpn = alpn;
      if (n.fp) proxy['client-fingerprint'] = n.fp;
      if (n.network && n.network !== 'tcp') {
        proxy.network = n.network === 'httpupgrade' ? 'ws' : n.network;
        applyTransport(proxy, n);
      }
      break;
    case 'ss': {
      proxy.type = 'ss';
      proxy.cipher = n.method;
      proxy.password = n.password;
      if (n.plugin) {
        const [pn, ...opts] = n.plugin.split(';');
        const o = {};
        opts.forEach((x) => {
          const [k, v] = splitFirst(x, '=');
          if (k) o[k] = v;
        });
        if (/obfs/.test(pn)) {
          proxy.plugin = 'obfs';
          proxy['plugin-opts'] = { mode: o.obfs || 'http', host: o['obfs-host'] || '' };
        } else if (pn.includes('v2ray-plugin')) {
          proxy.plugin = 'v2ray-plugin';
          proxy['plugin-opts'] = {
            mode: 'websocket',
            tls: 'tls' in o,
            host: o.host || '',
            path: o.path || '/',
            mux: 'mux' in o,
          };
        }
      }
      break;
    }
    case 'ssr':
      proxy.type = 'ssr';
      proxy.cipher = n.method;
      proxy.password = n.password;
      proxy.obfs = n.obfs;
      proxy.protocol = n.ssrProtocol;
      if (n.obfsParam) proxy['obfs-param'] = n.obfsParam;
      if (n.protocolParam) proxy['protocol-param'] = n.protocolParam;
      break;
    case 'hysteria2':
      proxy.type = 'hysteria2';
      proxy.password = n.password;
      proxy.sni = n.sni;
      proxy['skip-cert-verify'] = !!n.insecure;
      if (alpn.length) proxy.alpn = alpn;
      if (n.mport) proxy.ports = n.mport;
      if (n.obfs) {
        proxy.obfs = n.obfs;
        proxy['obfs-password'] = n.obfsPassword;
      }
      break;
    case 'hysteria':
      proxy.type = 'hysteria';
      proxy['auth-str'] = n.auth;
      proxy.sni = n.sni;
      proxy['skip-cert-verify'] = !!n.insecure;
      proxy.up = /^\d+$/.test(n.up) ? n.up + ' Mbps' : n.up;
      proxy.down = /^\d+$/.test(n.down) ? n.down + ' Mbps' : n.down;
      if (alpn.length) proxy.alpn = alpn;
      if (n.hyProtocol) proxy.protocol = n.hyProtocol;
      if (n.obfs) proxy.obfs = n.obfsParam || n.obfs;
      break;
    case 'tuic':
      proxy.type = 'tuic';
      proxy.uuid = n.uuid;
      proxy.password = n.password;
      proxy.sni = n.sni;
      proxy['congestion-controller'] = n.congestion;
      proxy['udp-relay-mode'] = n.udpRelayMode;
      proxy['skip-cert-verify'] = !!n.insecure;
      if (alpn.length) proxy.alpn = alpn;
      if (n.disableSni) proxy['disable-sni'] = true;
      break;
    case 'anytls':
      proxy.type = 'anytls';
      proxy.password = n.password;
      proxy.sni = n.sni;
      proxy['skip-cert-verify'] = !!n.insecure;
      if (alpn.length) proxy.alpn = alpn;
      if (n.fp) proxy['client-fingerprint'] = n.fp;
      break;
    case 'wireguard':
      proxy.type = 'wireguard';
      proxy['private-key'] = n.privateKey;
      proxy['public-key'] = n.publicKey;
      if (n.ip) proxy.ip = n.ip;
      if (n.ipv6) proxy.ipv6 = n.ipv6;
      if (n.presharedKey) proxy['pre-shared-key'] = n.presharedKey;
      if (n.reserved && n.reserved.length) proxy.reserved = n.reserved;
      if (n.mtu) proxy.mtu = n.mtu;
      break;
    case 'masque':
      proxy.type = 'masque';
      proxy['private-key'] = n.privateKey;
      proxy['public-key'] = n.publicKey;
      if (n.ip) proxy.ip = n.ip;
      if (n.ipv6) proxy.ipv6 = n.ipv6;
      if (n.uri) proxy.uri = n.uri;
      if (n.sni) proxy.sni = n.sni;
      if (n.mtu) proxy.mtu = n.mtu;
      break;
    case 'mieru':
      proxy.type = 'mieru';
      proxy.transport = n.transport || 'TCP';
      proxy.username = n.username;
      proxy.password = n.password;
      if (n.portRange) proxy['port-range'] = n.portRange;
      if (n.multiplexing) proxy.multiplexing = n.multiplexing;
      break;
    case 'socks5':
      proxy.type = 'socks5';
      if (n.username) proxy.username = n.username;
      if (n.password) proxy.password = n.password;
      break;
  }
  return proxy;
}

// ============================== 鉴权 ==============================

const SESSION_TTL = 7 * 24 * 3600; // 7 天

async function createSession(kv, username) {
  const sid = uid(32);
  await kv.put('session:' + sid, JSON.stringify({ username, exp: Date.now() + SESSION_TTL * 1000 }), {
    expirationTtl: SESSION_TTL,
  });
  return sid;
}

async function getSession(kv, req) {
  const cookies = parseCookies(req);
  const sid = cookies['sid'];
  if (!sid) return null;
  return await kv.get('session:' + sid, 'json');
}

async function requireAuth(kv, req) {
  const s = await getSession(kv, req);
  return !!s;
}

function setCookieHeader(sid, maxAge) {
  return `sid=${sid}; Path=/; HttpOnly; Secure; SameSite=Lax; Max-Age=${maxAge}`;
}

function clearCookieHeader() {
  return `sid=; Path=/; HttpOnly; Secure; SameSite=Lax; Max-Age=0`;
}

async function fetchSourceNodes(url) {
  // 使用 v2rayN 的 UA：多数机场会返回 Base64 链接列表，而不是 Clash YAML
  const res = await fetch(url, { headers: { 'User-Agent': 'v2rayN/7.0' } });
  if (!res.ok) throw new Error('源站请求失败: HTTP ' + res.status);
  const text = await res.text();
  const nodes = [];
  for (const line of extractNodeLinks(text)) {
    try {
      nodes.push(parseNodeUri(line));
    } catch (e) {
      // 跳过无法解析的行
    }
  }
  return nodes;
}

async function collectTokenNodes(kv, tokenRecord) {
  const allNodes = await getNodes(kv);
  const allSources = await getSources(kv);

  let manualNodes;
  if (tokenRecord.includeAllNodes) {
    manualNodes = allNodes.filter((n) => n.enabled !== false);
  } else {
    const idSet = new Set(tokenRecord.nodeIds || []);
    manualNodes = allNodes.filter((n) => idSet.has(n.id) && n.enabled !== false);
  }
  const manualParsed = [];
  for (const n of manualNodes) {
    try {
      const parsed = parseNodeUri(n.raw);
      if (n.name) parsed.name = n.name; // 使用控制台中保存/修改后的名称
      manualParsed.push(parsed);
    } catch (e) {
      /* 忽略损坏节点 */
    }
  }

  let sources;
  if (tokenRecord.includeAllSources) {
    sources = allSources.filter((s) => s.enabled !== false);
  } else {
    const idSet = new Set(tokenRecord.sourceIds || []);
    sources = allSources.filter((s) => idSet.has(s.id) && s.enabled !== false);
  }

  const remoteResults = await Promise.allSettled(sources.map((s) => fetchSourceNodes(s.url)));
  const remoteParsed = [];
  remoteResults.forEach((r) => {
    if (r.status === 'fulfilled') remoteParsed.push(...r.value);
  });

  // 去重
  const seen = new Set();
  const merged = [];
  for (const n of [...manualParsed, ...remoteParsed]) {
    const key = [
      n.protocol,
      n.server,
      n.port,
      n.uuid || n.password || n.privateKey || n.auth || n.username || '',
    ].join('|');
    if (seen.has(key)) continue;
    seen.add(key);
    merged.push(n);
  }
  return merged;
}

function buildClashYaml(allNodes, tokenRecord) {
  // Clash 无法表示 naive / juicity，跳过
  const nodes = allNodes.filter((n) => !CLASH_UNSUPPORTED.has(n.protocol));
  const skipped = allNodes.length - nodes.length;

  const usedNames = new Set();
  nodes.forEach((n) => {
    n.name = uniqueName(n.name || n.server, usedNames);
  });
  const proxies = nodes.map(nodeToClashProxy);
  const proxyNames = proxies.map((p) => p.name);

  const config = {
    'mixed-port': 7890,
    'allow-lan': true,
    mode: 'rule',
    'log-level': 'info',
    'external-controller': '127.0.0.1:9090',
    dns: {

      'default-nameserver': ['223.5.5.5', '119.29.29.29'],
      nameserver: ['https://doh.pub/dns-query', 'https://dns.alidns.com/dns-query'],
    },
    proxies,
    'proxy-groups': [
      {
        name: PROXY_GROUP_MAIN,
        type: 'select',
        proxies: [PROXY_GROUP_AUTO, 'DIRECT', ...proxyNames],
      },
      {
        name: PROXY_GROUP_AUTO,
        type: 'url-test',
        proxies: proxyNames.length ? proxyNames : ['DIRECT'],
        url: 'https://www.gstatic.com/generate_204',
        interval: 300,
        tolerance: 50,
      },
    ],
    'rule-providers': {
      GoogleCNProxyIP: ruleProvider('GoogleCNProxyIP'),
      LocalAreaNetwork: ruleProvider('LocalAreaNetwork'),
      UnBan: ruleProvider('UnBan'),
      ChinaDomain: ruleProvider('ChinaDomain'),
      ChinaMedia: ruleProvider('ChinaMedia'),
      BanAD: ruleProvider('BanAD'),
      BanProgramAD: ruleProvider('BanProgramAD'),
      ChinaCompanyIp: ruleProvider('ChinaCompanyIp'),
      ChinaIp: ruleProvider('ChinaIp'),
    },
    rules: [
      `DOMAIN-SUFFIX,xn--ngstr-lra8j.com,${PROXY_GROUP_MAIN}`,
      `DOMAIN-SUFFIX,services.googleapis.cn,${PROXY_GROUP_MAIN}`,
      `RULE-SET,GoogleCNProxyIP,${PROXY_GROUP_MAIN}`,
      'RULE-SET,LocalAreaNetwork,DIRECT',

      'RULE-SET,ChinaDomain,DIRECT',

    ],
  };

  let yaml = '# 聚合订阅 - 自动生成\n';
  yaml += `# 名称: ${safeName}\n`;
  if (skipped) yaml += `# 已跳过 ${skipped} 个 Clash 不支持的节点(naive/juicity)\n`;
  yaml += yamlDump(config, 0);
  return yaml;
}

function buildRawSubscription(nodes) {
  const lines = nodes.map((n) => n.raw);
  return b64EncodeUtf8(lines.join('\n'));
}

function subUserInfoHeader(meta) {
  meta = meta || {};

  const parts = [`upload=${upload}`, `download=${download}`, `total=${total}`];
  if (meta.expire) parts.push(`expire=${meta.expire}`);
  return parts.join('; ');
}

// ============================== API 路由处理 ==============================

async function handleApi(req, env, url) {
  const kv = env.SUB_KV;
  const method = req.method;
  const path = url.pathname;

  if (path === '/api/login' && method === 'POST') {
    const body = await req.json().catch(() => ({}));
    const { username, password } = body;
    if (
      username === env.ADMIN_USERNAME &&
      password === env.ADMIN_PASSWORD &&
      env.ADMIN_USERNAME &&
      env.ADMIN_PASSWORD
    ) {
      const sid = await createSession(kv, username);
      return jsonResponse({ ok: true }, 200, { 'Set-Cookie': setCookieHeader(sid, SESSION_TTL) });
    }
    return jsonResponse({ ok: false, error: '用户名或密码错误' }, 401);
  }

  if (path === '/api/logout' && method === 'POST') {
    const cookies = parseCookies(req);
    if (cookies['sid']) await kv.delete('session:' + cookies['sid']);
    return jsonResponse({ ok: true }, 200, { 'Set-Cookie': clearCookieHeader() });
  }

  // 以下接口需要登录
  const authed = await requireAuth(kv, req);
  if (!authed) return jsonResponse({ ok: false, error: '未登录' }, 401);

  // ---- 节点 ----
  if (path === '/api/nodes' && method === 'GET') {
    const nodes = await getNodes(kv);
    return jsonResponse({ ok: true, data: nodes });
  }
  if (path === '/api/nodes' && method === 'POST') {
    const body = await req.json().catch(() => ({}));
    const lines = extractNodeLinks(body.raw);
    if (!lines.length) return jsonResponse({ ok: false, error: '未识别到有效的节点链接' }, 400);
    const nodes = await getNodes(kv);
    const added = [];
    const failed = [];
    for (const line of lines) {
      try {
        const parsed = parseNodeUri(line);
        const record = {
          id: uid(10),
          raw: line,
          protocol: parsed.protocol,
          name: parsed.name,
          server: parsed.server,
          port: parsed.port,
          enabled: true,
          addedAt: Date.now(),
        };
        nodes.push(record);
        added.push(record);
      } catch (e) {
        failed.push({ line, error: e.message });
      }
    }
    await saveNodes(kv, nodes);
    return jsonResponse({ ok: true, added, failed });
  }
  {
    const m = path.match(/^\/api\/nodes\/([^/]+)$/);
    if (m && (method === 'PUT' || method === 'DELETE')) {
      const id = m[1];
      let nodes = await getNodes(kv);
      if (method === 'DELETE') {
        nodes = nodes.filter((n) => n.id !== id);
        await saveNodes(kv, nodes);
        return jsonResponse({ ok: true });
      }
      const body = await req.json().catch(() => ({}));
      const idx = nodes.findIndex((n) => n.id === id);
      if (idx < 0) return jsonResponse({ ok: false, error: '节点不存在' }, 404);

      // 二次修改节点链接本身（raw），重新解析并刷新 protocol/server/port/name
      if (typeof body.raw === 'string' && body.raw.trim()) {
        const links = extractNodeLinks(body.raw);
        if (links.length !== 1) {
          return jsonResponse({ ok: false, error: '编辑时需要且只能包含一条有效的节点链接' }, 400);
        }
        let parsed;
        try {
          parsed = parseNodeUri(links[0]);
        } catch (e) {
          return jsonResponse({ ok: false, error: '节点链接解析失败: ' + e.message }, 400);
        }
        nodes[idx].raw = links[0];
        nodes[idx].protocol = parsed.protocol;
        nodes[idx].server = parsed.server;
        nodes[idx].port = parsed.port;
        nodes[idx].name = parsed.name;
      }
      // 显式传入的 name 会覆盖从链接中解析出的名称
      if (typeof body.name === 'string' && body.name.trim()) nodes[idx].name = body.name.trim();
      if (typeof body.enabled === 'boolean') nodes[idx].enabled = body.enabled;
      await saveNodes(kv, nodes);
      return jsonResponse({ ok: true, data: nodes[idx] });
    }
  }

  // ---- 订阅源 ----
  if (path === '/api/sources' && method === 'GET') {
    return jsonResponse({ ok: true, data: await getSources(kv) });
  }
  if (path === '/api/sources' && method === 'POST') {
    const body = await req.json().catch(() => ({}));
    if (!body.url) return jsonResponse({ ok: false, error: '缺少 url' }, 400);
    const sources = await getSources(kv);
    const record = { id: uid(10), url: body.url, name: body.name || body.url, enabled: true, addedAt: Date.now() };
    sources.push(record);
    await saveSources(kv, sources);
    return jsonResponse({ ok: true, data: record });
  }
  {
    const m = path.match(/^\/api\/sources\/([^/]+)$/);
    if (m && (method === 'PUT' || method === 'DELETE')) {
      const id = m[1];
      let sources = await getSources(kv);
      if (method === 'DELETE') {
        sources = sources.filter((s) => s.id !== id);
        await saveSources(kv, sources);
        return jsonResponse({ ok: true });
      }
      const body = await req.json().catch(() => ({}));
      const idx = sources.findIndex((s) => s.id === id);
      if (idx < 0) return jsonResponse({ ok: false, error: '订阅源不存在' }, 404);
      if (typeof body.name === 'string') sources[idx].name = body.name;
      if (typeof body.enabled === 'boolean') sources[idx].enabled = body.enabled;
      if (typeof body.url === 'string') sources[idx].url = body.url;
      await saveSources(kv, sources);
      return jsonResponse({ ok: true, data: sources[idx] });
    }
  }

  // ---- 订阅口令 ----
  if (path === '/api/tokens' && method === 'GET') {
    return jsonResponse({ ok: true, data: await getTokens(kv) });
  }
  if (path === '/api/tokens' && method === 'POST') {
    const body = await req.json().catch(() => ({}));
    const tokens = await getTokens(kv);
    const record = {
      token: randomToken(28),
      name: body.name || '未命名订阅',
      nodeIds: Array.isArray(body.nodeIds) ? body.nodeIds : [],
      sourceIds: Array.isArray(body.sourceIds) ? body.sourceIds : [],
      includeAllNodes: !!body.includeAllNodes,
      includeAllSources: !!body.includeAllSources,
      enabled: true,
      meta: {
        total: Number(body.meta?.total) || 0,
        upload: Number(body.meta?.upload) || 0,
        download: Number(body.meta?.download) || 0,
        expire: body.meta?.expire ? Number(body.meta.expire) : 0,
      },
      createdAt: Date.now(),
    };
    tokens.push(record);
    await saveTokens(kv, tokens);
    return jsonResponse({ ok: true, data: record });
  }
  {
    const m = path.match(/^\/api\/tokens\/([^/]+)$/);
    if (m && (method === 'PUT' || method === 'DELETE')) {
      const token = m[1];
      let tokens = await getTokens(kv);
      if (method === 'DELETE') {
        tokens = tokens.filter((t) => t.token !== token);
        await saveTokens(kv, tokens);
        return jsonResponse({ ok: true });
      }
      const body = await req.json().catch(() => ({}));
      const idx = tokens.findIndex((t) => t.token === token);
      if (idx < 0) return jsonResponse({ ok: false, error: '订阅不存在' }, 404);
      const t = tokens[idx];
      if (typeof body.name === 'string') t.name = body.name;
      if (typeof body.enabled === 'boolean') t.enabled = body.enabled;
      if (Array.isArray(body.nodeIds)) t.nodeIds = body.nodeIds;
      if (Array.isArray(body.sourceIds)) t.sourceIds = body.sourceIds;
      if (typeof body.includeAllNodes === 'boolean') t.includeAllNodes = body.includeAllNodes;
      if (typeof body.includeAllSources === 'boolean') t.includeAllSources = body.includeAllSources;
      if (body.meta) {
        t.meta = {
          total: Number(body.meta.total) || 0,
          upload: Number(body.meta.upload) || 0,
          download: Number(body.meta.download) || 0,
          expire: body.meta.expire ? Number(body.meta.expire) : 0,
        };
      }
      await saveTokens(kv, tokens);
      return jsonResponse({ ok: true, data: t });
    }
  }

  return jsonResponse({ ok: false, error: 'Not Found' }, 404);
}

// ============================== 订阅输出 ==============================

async function handleSub(req, env, url) {
  const kv = env.SUB_KV;
  const token = url.pathname.replace(/^\/sub\//, '');
  const tokens = await getTokens(kv);
  const record = tokens.find((t) => t.token === token);
  if (!record || record.enabled === false) {
    return new Response('订阅不存在或已停用', { status: 404 });
  }

  let nodes;
  try {
    nodes = await collectTokenNodes(kv, record);
  } catch (e) {
    return new Response('聚合节点失败: ' + e.message, { status: 502 });
  }

  const ua = req.headers.get('User-Agent') || '';
  const target = url.searchParams.get('target'); // clash | raw
  const useClash = target === 'clash' || (target !== 'raw' && isClashUA(ua));

  const headers = {
    'Subscription-Userinfo': subUserInfoHeader(record.meta),
    'Cache-Control': 'no-store',
  };

  if (useClash) {
    const yaml = buildClashYaml(nodes, record);
    headers['Content-Type'] = 'text/yaml; charset=utf-8';
    headers['Content-Disposition'] = `attachment; filename="${encodeURIComponent(record.name)}.yaml"`;
    return new Response(yaml, { status: 200, headers });
  } else {
    const b64 = buildRawSubscription(nodes);
    headers['Content-Type'] = 'text/plain; charset=utf-8';
    headers['Content-Disposition'] = `attachment; filename="${encodeURIComponent(record.name)}.txt"`;
    return new Response(b64, { status: 200, headers });
  }
}

export default {
  async fetch(req, env) {
    const url = new URL(req.url);
    const path = url.pathname;

    try {
      if (path.startsWith('/api/')) {
        return await handleApi(req, env, url);
      }
      if (path.startsWith('/sub/')) {
        return await handleSub(req, env, url);
      }
      if (path === '/login') {
        return new Response(LOGIN_HTML, { headers: { 'Content-Type': 'text/html; charset=utf-8' } });
      }
      // 根路径与其他路径：需要登录才能看到控制台
      const authed = await requireAuth(env.SUB_KV, req);
      if (!authed) {
        return Response.redirect(url.origin + '/login', 302);
      }
      return new Response(PORTAL_HTML, { headers: { 'Content-Type': 'text/html; charset=utf-8' } });
    } catch (err) {
      return new Response('服务器错误: ' + err.message, { status: 500 });
    }
  },
};
